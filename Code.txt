    box.Size = UDim2.new(0, 65, 1, 0); box.Position = UDim2.new(0, 95, 0, 0)
    box.BackgroundColor3 = Color3.fromRGB(25, 25, 25); box.BorderColor3 = Color3.fromRGB(50, 50, 50)
    box.Text = tostring(defaultVal); box.TextColor3 = Color3.fromRGB(255, 255, 255)
    box.Font = Enum.Font.SourceSansBold; box.TextSize = 12; box.ClearTextOnFocus = false; box.Parent = container

    box.FocusLost:Connect(function()
        local num = tonumber(box.Text)
        if num then callback(num) else box.Text = tostring(defaultVal) end
    end)
    return box
end

-- Chat Helper Function (Legacy Chat & TextChatService support)
local function sendChatMessage(msg)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local generalChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
            if generalChannel then
                generalChannel:SendAsync(msg)
            end
        else
            ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All")
        end
    end)
end

local function refreshJumpscareList()
    for _, child in ipairs(JumpscarePanel:GetChildren()) do if child:IsA("TextButton") then child:Destroy() end end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local pBtn = Instance.new("TextButton")
            pBtn.Size = UDim2.new(0, 135, 0, 24); pBtn.BackgroundColor3 = Color3.fromRGB(40, 20, 20); pBtn.BorderSizePixel = 0
            pBtn.Text = "👻 " .. p.DisplayName; pBtn.TextColor3 = Color3.fromRGB(255, 100, 100); pBtn.Font = Enum.Font.SourceSansBold; pBtn.TextSize = 12; pBtn.Parent = JumpscarePanel
            
            pBtn.MouseButton1Click:Connect(function()
                if isJumpscaring then return end
                
                local myChar = LocalPlayer.Character
                local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
                local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
                local targetHRP = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                
                if myHRP and targetHRP and myHum then
                    isJumpscaring = true
                    local originalCFrame = myHRP.CFrame
                    sendChatMessage("BOO!")
                    
                    local startTime = tick()
                    local offsets = {
                        Vector3.new(0, 0, -2.5), Vector3.new(0, 0, 2.5),
                        Vector3.new(-2.5, 0, 0), Vector3.new(2.5, 0, 0),
                        Vector3.new(0, 2, 0)
                    }
                    
                    task.spawn(function()
                        local idx = 1
                        while tick() - startTime < 1.0 do
                            if p.Character and p.Character:FindFirstChild("HumanoidRootPart") and myHRP then
                                local targetPos = p.Character.HumanoidRootPart.Position
                                local currentOffset = offsets[(idx % #offsets) + 1]
                                myHRP.AssemblyLinearVelocity = Vector3.zero
                                myHRP.AssemblyAngularVelocity = Vector3.zero
                                myHRP.CFrame = CFrame.new(targetPos + currentOffset, targetPos)
                                idx = idx + 1
                            end
                            task.wait(0.03)
                        end
                        
                        -- Unfreeze / Movement Restoration Logic
                        if myHRP then
                            myHRP.AssemblyLinearVelocity = Vector3.zero
                            myHRP.AssemblyAngularVelocity = Vector3.zero
                            myHRP.CFrame = originalCFrame
                        end
                        if myHum then
                            myHum.PlatformStand = false
                            myHum:ChangeState(Enum.HumanoidStateType.GettingUp)
                            myHum.WalkSpeed = walkSpeed
                            myHum.JumpPower = jumpPower
                        end
                        isJumpscaring = false
                    end)
                end
            end)
        end
    end
end

        Lighting.Ambient = Color3.fromRGB(0, 0, 0)
    end
end)

local function disableAllFeatures()
    if BackgroundSound then BackgroundSound:Stop() end

    flying = false
    if flyKeepAliveConnection then flyKeepAliveConnection:Disconnect(); flyKeepAliveConnection = nil end
    if flyBV then flyBV:Destroy(); flyBV = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end

    autoTargetActive = false
    TargetStatusPanel.Visible = false
    if autoTargetConnection then autoTargetConnection:Disconnect(); autoTargetConnection = nil end
    if autoTargetInputConnection then autoTargetInputConnection:Disconnect(); autoTargetInputConnection = nil end

    waypointsActive = false
    WaypointPanel.Visible = false
    if waypointInputConnection then waypointInputConnection:Disconnect(); waypointInputConnection = nil end

    WeaponsPanel.Visible = false
    GoofyPanel.Visible = false
    JumpscarePanel.Visible = false

    if discoActive then
        discoActive = false
        if discoThread then task.cancel(discoThread) end
        Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
        Lighting.Ambient = Color3.fromRGB(0, 0, 0)
    end

    espActive = false
    if espConnection then espConnection:Disconnect(); espConnection = nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            if p.Character:FindFirstChild("ESPHighlight") then p.Character.ESPHighlight:Destroy() end
            if p.Character:FindFirstChild("Head") and p.Character.Head:FindFirstChild("ESPNameTag") then
                p.Character.Head.ESPNameTag:Destroy()
            end
        end
    end

    noclip = false
    if noclipConnection then noclipConnection:Disconnect(); noclipConnection = nil end
    if noclipKeepAliveConnection then noclipKeepAliveConnection:Disconnect(); noclipKeepAliveConnection = nil end

    fullBright = false
    Lighting.Brightness = 1; Lighting.ClockTime = 12; Lighting.GlobalShadows = true

    noFog = false
    Lighting.FogEnd = 1000

    infJump = false
    if infJumpConnection then infJumpConnection:Disconnect(); infJumpConnection = nil end

    if walkSpeedConnection then walkSpeedConnection:Disconnect(); walkSpeedConnection = nil end

    if LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
        local hum = LocalPlayer.Character:FindFirstChildOfClass("Humanoid")
        hum.WalkSpeed = 16
        hum.JumpPower = 50
    end
end

local function giveWeaponToPlayer(toolObj)
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        local clonedTool = toolObj:Clone()
        clonedTool.Parent = backpack
    end

    pcall(function()
        for _, desc in ipairs(toolObj:GetDescendants()) do
            if desc:IsA("RemoteEvent") then
                desc:FireServer()
            elseif desc:IsA("RemoteFunction") then
                desc:InvokeServer()
            end
        end
    end)

    pcall(function()
        for _, res in ipairs(ReplicatedStorage:GetDescendants()) do
            if res:IsA("RemoteEvent") then
                local lowerName = res.Name:lower()
                if lowerName:find("give") or lowerName:find("buy") or lowerName:find("equip") or lowerName:find("weapon") then
                    res:FireServer(toolObj.Name)
                    res:FireServer(toolObj)
                end
            end
        end
    end)
end

local function refreshWeaponsList()
    for _, child in ipairs(WeaponsScroll:GetChildren()) do
        if child:IsA("TextButton") then child:Destroy() end
    end

    local foundWeapons = {}

    local function searchContainer(container)
        for _, item in ipairs(container:GetDescendants()) do
            if item:IsA("Tool") and not foundWeapons[item.Name] then
                foundWeapons[item.Name] = item
            end
        end
    end

    searchContainer(Lighting)
    searchContainer(ReplicatedStorage)
    searchContainer(workspace)
    for _, p in ipairs(Players:GetPlayers()) do
        if p:FindFirstChild("Backpack") then searchContainer(p.Backpack) end
    end

    for name, toolObj in pairs(foundWeapons) do
        local wBtn = Instance.new("TextButton")
        wBtn.Size = UDim2.new(1, 0, 0, 22)
        wBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
        wBtn.BorderSizePixel = 0
        wBtn.Text = name
        wBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
        wBtn.Font = Enum.Font.SourceSansBold
        wBtn.TextSize = 12
        wBtn.Parent = WeaponsScroll

        wBtn.MouseButton1Click:Connect(function()
            giveWeaponToPlayer(toolObj)
        end)
    end
end

local function getPlayerUnderCursor()
    local mousePos = UserInputService:GetMouseLocation()
    local unitRay = Camera:ViewportPointToRay(mousePos.X, mousePos.Y)
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude

    local ignoreList = {}
    if LocalPlayer.Character then table.insert(ignoreList, LocalPlayer.Character) end
    raycastParams.FilterDescendantsInstances = ignoreList

    local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * 1000, raycastParams)
    if result and result.Instance then
        local hitChar = result.Instance:FindFirstAncestorOfClass("Model")
        if hitChar then
            local player = Players:GetPlayerFromCharacter(hitChar)
            if player and player ~= LocalPlayer then return player end
        end
    end
    return nil
end

-- Returns true when the camera has a clear line of sight to ANY usable limb/body part.
-- The target is considered visible if at least one ray reaches the target character.
local function canSeeTarget(p)
    if not p or not p.Character then return false end

    local targetChar = p.Character
    local myChar = LocalPlayer.Character
    if not myChar then return false end

    local origin = Camera.CFrame.Position
    local visibleParts = {
        "Head",
        "UpperTorso",
        "LowerTorso",
        "Torso",
        "HumanoidRootPart",
        "LeftUpperArm",
        "LeftLowerArm",
        "LeftHand",
        "RightUpperArm",
        "RightLowerArm",
        "RightHand",
        "LeftUpperLeg",
        "LeftLowerLeg",
        "LeftFoot",
        "RightUpperLeg",
        "RightLowerLeg",
        "RightFoot"
    }

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {myChar}

    for _, partName in ipairs(visibleParts) do
        local part = targetChar:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            local direction = part.Position - origin
            local result = workspace:Raycast(origin, direction, raycastParams)

            if result and result.Instance and result.Instance:IsDescendantOf(targetChar) then
                return true
            end
        end
    end

    return false
end

local function getClosestPlayer()
    local closest, shortestDist = nil, math.huge
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer
            and not markedPlayers[p]
            and p.Character
            and p.Character:FindFirstChild("HumanoidRootPart")
            and p.Character:FindFirstChildOfClass("Humanoid")
            and p.Character:FindFirstChildOfClass("Humanoid").Health > 0
            and canSeeTarget(p) then

            local targetPos = p.Character.HumanoidRootPart.Position
            local dist = (myPos - targetPos).Magnitude

            if dist < shortestDist then
                shortestDist = dist
                closest = p
            end
        end
    end

    return closest
end

-- Flight System
local function ensureFlyObjects()
    if not flying then return false end

    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then
        if flyBV then flyBV:Destroy(); flyBV = nil end
        if flyBG then flyBG:Destroy(); flyBG = nil end
        return false
    end

    if not flyBV or flyBV.Parent ~= hrp then
        if flyBV then flyBV:Destroy() end
        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        flyBV.Parent = hrp
    else
        flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
    end

    if not flyBG or flyBG.Parent ~= hrp then
        if flyBG then flyBG:Destroy() end
        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        flyBG.Parent = hrp
    else
        flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    end

    return true
end
