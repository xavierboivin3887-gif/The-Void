-- Chat Helper Function (Legacy Chat & TextChatService support)
local function sendChatMessage(msg)
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local generalChannel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
            if generalChannel then
                generalChannel:SendAsync(msg)
            end
        else
            ReplicatedStorage.DefaultChatSystemChatEvents.SayMessageRequest:FireServer(msg, "All")
        end
    end)
end

local function playMusicId(id)
    if currentSound then
        currentSound:Stop()
        currentSound:Destroy()
        currentSound = nil
    end
    
    local soundId = tonumber(id) and ("rbxassetid://" .. tostring(id)) or tostring(id)
    
    local sound = Instance.new("Sound")
    sound.SoundId = soundId
    sound.Volume = 1
    sound.Looped = true
    sound.Parent = SoundService or Camera or LocalPlayer:WaitForChild("PlayerGui")
    
    currentSound = sound
    sound:Play()
end

local function refreshJumpscareList()
    for _, child in ipairs(JumpscarePanel:GetChildren()) do if child:IsA("TextButton") then child:Destroy() end end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local pBtn = Instance.new("TextButton")
            pBtn.Size = UDim2.new(0, 135, 0, 24); pBtn.BackgroundColor3 = Color3.fromRGB(40, 20, 20); pBtn.BorderSizePixel = 0
            pBtn.Text = "👻 " .. p.DisplayName; pBtn.TextColor3 = Color3.fromRGB(255, 100, 100); pBtn.Font = Enum.Font.SourceSansBold; pBtn.TextSize = 12; pBtn.Parent = JumpscarePanel
            
            pBtn.MouseButton1Click:Connect(function()
                if isJumpscaring then return end
                
                local myChar = LocalPlayer.Character
                local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
                local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
                local targetHRP = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                
                if myHRP and targetHRP and myHum then
                    isJumpscaring = true
                    local originalCFrame = myHRP.CFrame
                    sendChatMessage("BOO!")
                    
                    local startTime = tick()
                    local offsets = {
                        Vector3.new(0, 0, -2.5), Vector3.new(0, 0, 2.5),
                        Vector3.new(-2.5, 0, 0), Vector3.new(2.5, 0, 0),
                        Vector3.new(0, 2, 0)
                    }
                    
                    task.spawn(function()
                        local idx = 1
                        while tick() - startTime < 1.0 do
                            if p.Character and p.Character:FindFirstChild("HumanoidRootPart") and myHRP then
                                local targetPos = p.Character.HumanoidRootPart.Position
                                local currentOffset = offsets[(idx % #offsets) + 1]
                                myHRP.AssemblyLinearVelocity = Vector3.zero
                                myHRP.AssemblyAngularVelocity = Vector3.zero
                                myHRP.CFrame = CFrame.new(targetPos + currentOffset, targetPos)
                                idx = idx + 1
                            end
                            task.wait(0.03)
                        end
                        
                        -- Unfreeze / Movement Restoration Logic
                        if myHRP then
                            myHRP.AssemblyLinearVelocity = Vector3.zero
                            myHRP.AssemblyAngularVelocity = Vector3.zero
                            myHRP.CFrame = originalCFrame
                        end
                        if myHum then
                            myHum.PlatformStand = false
                            myHum:ChangeState(Enum.HumanoidStateType.GettingUp)
                            myHum.WalkSpeed = walkSpeed
                            myHum.JumpPower = jumpPower
                        end
                        isJumpscaring = false
                    end)
                end
            end)
        end
    end
end

JumpscareBtn.MouseButton1Click:Connect(function()

    if JumpscarePanel.Visible then refreshJumpscareList() end
end)

local function getAttackerPlayer()
    local closest, shortestDist = nil, math.huge
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local dist = (myPos - p.Character.HumanoidRootPart.Position).Magnitude

local function setupAutoDodge()
    if autoDodgeConnection then autoDodgeConnection:Disconnect(); autoDodgeConnection = nil end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not hum then return end

    local lastHealth = hum.Health

    autoDodgeConnection = hum.HealthChanged:Connect(function(newHealth)
        if newHealth < lastHealth and newHealth > 0 then
            local myChar = LocalPlayer.Character
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHRP then
                if autoDodgeMode == "bullet" then
                    local originalCFrame = myHRP.CFrame
                    local offsets = {
                        Vector3.new(0, 0, -5), Vector3.new(0, 0, 5),
                        Vector3.new(-5, 0, 0), Vector3.new(5, 0, 0),
                        Vector3.new(0, 4, 0)
                    }
                    task.spawn(function()
                        local startTime = tick()
                        local idx = 1
                        while tick() - startTime < 0.5 do
                            if myHRP then
                                local currentOffset = offsets[(idx % #offsets) + 1]
                                myHRP.AssemblyLinearVelocity = Vector3.zero
                                myHRP.AssemblyAngularVelocity = Vector3.zero
                                myHRP.CFrame = originalCFrame * CFrame.new(currentOffset)
                                idx = idx + 1
                            end
                            task.wait(0.03)
                        end
                        if myHRP then
                            myHRP.AssemblyLinearVelocity = Vector3.zero
                            myHRP.AssemblyAngularVelocity = Vector3.zero
                            myHRP.CFrame = originalCFrame
                        end
                    end)
                elseif autoDodgeMode == "sword" then
                    local attacker = getAttackerPlayer()
                    if attacker and attacker.Character and attacker.Character:FindFirstChild("HumanoidRootPart") then
                        local attackerHRP = attacker.Character.HumanoidRootPart
                        myHRP.AssemblyLinearVelocity = Vector3.zero
                        myHRP.AssemblyAngularVelocity = Vector3.zero
                        myHRP.CFrame = attackerHRP.CFrame * CFrame.new(0, 0, 3)
                        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, true, game, 0)
                        VirtualInputManager:SendMouseButtonEvent(0, 0, 0, false, game, 0)
                    end
                end
            end
        end
        lastHealth = newHealth
    end)
end

local function getPlayerUnderCursor()
    local mousePos = UserInputService:GetMouseLocation()
    local unitRay = Camera:ViewportPointToRay(mousePos.X, mousePos.Y)
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude

    local ignoreList = {}
    if LocalPlayer.Character then table.insert(ignoreList, LocalPlayer.Character) end
    raycastParams.FilterDescendantsInstances = ignoreList

    local result = workspace:Raycast(unitRay.Origin, unitRay.Direction * 1000, raycastParams)
    if result and result.Instance then
        local hitChar = result.Instance:FindFirstAncestorOfClass("Model")
        if hitChar then
            local player = Players:GetPlayerFromCharacter(hitChar)
            if player and player ~= LocalPlayer then return player end
        end
    end
    return nil
end

-- Returns true when the camera has a clear line of sight to ANY usable limb/body part.
-- The target is considered visible if at least one ray reaches the target character.
local function canSeeTarget(p)
    if not p or not p.Character then return false end

    local targetChar = p.Character
    local myChar = LocalPlayer.Character
    if not myChar then return false end

    local origin = Camera.CFrame.Position
    local visibleParts = {
        "Head",
        "UpperTorso",
        "LowerTorso",
        "Torso",
        "HumanoidRootPart",
        "LeftUpperArm",
        "LeftLowerArm",
        "LeftHand",
        "RightUpperArm",
        "RightLowerArm",
        "RightHand",
        "LeftUpperLeg",
        "LeftLowerLeg",
        "LeftFoot",
        "RightUpperLeg",
        "RightLowerLeg",
        "RightFoot"
    }

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {myChar}

    for _, partName in ipairs(visibleParts) do
        local part = targetChar:FindFirstChild(partName)
        if part and part:IsA("BasePart") then
            local direction = part.Position - origin
            local result = workspace:Raycast(origin, direction, raycastParams)

            if result and result.Instance and result.Instance:IsDescendantOf(targetChar) then
                return true
            end
        end
    end

    return false
end

local function getClosestPlayer()
    local closest, shortestDist = nil, math.huge
    local myChar = LocalPlayer.Character
    if not myChar or not myChar:FindFirstChild("HumanoidRootPart") then return nil end
    local myPos = myChar.HumanoidRootPart.Position

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer
            and not markedPlayers[p]
            and p.Character
            and p.Character:FindFirstChild("HumanoidRootPart")
            and p.Character:FindFirstChildOfClass("Humanoid")
            and p.Character:FindFirstChildOfClass("Humanoid").Health > 0
            and canSeeTarget(p) then

            local targetPos = p.Character.HumanoidRootPart.Position
            local dist = (myPos - targetPos).Magnitude

            if dist < shortestDist then
                shortestDist = dist
                closest = p
            end
        end
    end

    return closest
end

local function ensureFlyObjects()
    if not flying then return false end

    local char = LocalPlayer.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hrp then
        if flyBV then flyBV:Destroy(); flyBV = nil end
        if flyBG then flyBG:Destroy(); flyBG = nil end
        return false
    end

    if not flyBV or flyBV.Parent ~= hrp then
        if flyBV then flyBV:Destroy() end
        flyBV = Instance.new("BodyVelocity")
        flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)
        flyBV.Parent = hrp
    else
        flyBV.MaxForce = Vector3.new(1e9, 1e9, 1e9)

    if not flyBG or flyBG.Parent ~= hrp then
        if flyBG then flyBG:Destroy() end
        flyBG = Instance.new("BodyGyro")
        flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        flyBG.Parent = hrp
    else
        flyBG.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
    end

    return true
end

InfJumpBtn.MouseButton1Click:Connect(function()
    infJump = not infJump
    InfJumpBtn.Text = infJump and "Infinite Jump: ON" or "Infinite Jump: OFF"
    InfJumpBtn.TextColor3 = infJump and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)

    if infJump then
        infJumpConnection = UserInputService.JumpRequest:Connect(function()
            if infJump and LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
                LocalPlayer.Character:FindFirstChildOfClass("Humanoid"):ChangeState(Enum.HumanoidStateType.Jumping)
            end
        end)
    else
        if infJumpConnection then infJumpConnection:Disconnect(); infJumpConnection = nil end
    end
end)

-- No Fog System
NoFogBtn.MouseButton1Click:Connect(function()
    noFog = not noFog
    NoFogBtn.Text = noFog and "No Fog: ON" or "No Fog: OFF"
    NoFogBtn.TextColor3 = noFog and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(255, 0, 0)
    Lighting.FogEnd = noFog and 1e5 or 1000
end)
